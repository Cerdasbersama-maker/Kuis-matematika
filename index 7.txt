<script>
function cekJawaban() {
    const pilihan = document.querySelector('input[name="answer"]:checked');
    const feedback = document.getElementById('feedback');

    if (!pilihan) {
        alert('Pilih salah satu jawaban!');
        return;
    }

    if (pilihan.value === 'D') {
        feedback.innerHTML = `<p>Jawaban anda <strong>benar</strong>.</p>`;
        feedback.style.backgroundColor = '#e0f7e9';
    } else {
        pilihan.parentElement.innerHTML += ' <span class="wrong">&#10006;</span>';
        feedback.innerHTML = `
            <p>Jawaban anda tidak benar.</p>
            <a href="#" class="button-pdf">Kunci Jawaban PDF</a>
            <p>Jawaban yang benar adalah: <strong>D. HCl dan NaOH</strong></p>
        `;
        feedback.classList.add('salah');
        feedback.style.display = 'block';
    }
    feedback.style.display = 'block';
}
</script>
